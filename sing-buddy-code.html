Wireless receiver and note playback

Starts ESP-NOW, sets each tone’s frequency and duration, and checks the buttons after receiving a message.

wlan = network.WLAN(network.STA_IF)
wlan.active(True)
e = espnow.ESPNow()
e.active(True)
print("ESP-NOW receiver ready...")

def play(notes):
    for note in notes:
        speaker.freq(note)
        speaker.duty(512)
        sleep(0.5)
        speaker.duty(0)
        sleep(0.1)

while True:
    host, msg = e.recv()
    if msg:
        try:
            decoded = msg.decode()
        except:
            decoded = str(msg)
        print("Received from", host, ":", decoded)
        play(music)
        if button1.value() == 0:
            play(low_notes)
        elif button2.value() == 0:
            play(mid_notes)
        elif button3.value() == 0:
            play(high_notes)
        else:
            speaker.duty(0)
        sleep(0.1)

RFID reader and LED

Radio, card reader, and status LED

Registers the receiver as an ESP-NOW peer, initializes the MFRC522 reader over SPI, and sets the scan LED low.

import network
import espnow
import time
from mfrc522 import MFRC522
from time import sleep
from machine import SPI, Pin

sta = network.WLAN(network.STA_IF)
sta.active(True)
e = espnow.ESPNow()
e.active(True)
peer = b'\x14\x2b\x2f\xaf\xf2\xa0'
e.add_peer(peer)
e.send(peer, b"Starting...")
spi = SPI(1, baudrate=1000000, polarity=0, phase=0)
rdr = MFRC522(spi, 27, 4)
led = Pin(15, Pin.OUT)
led.value(0)
print("Ready to scan...")
Read and transmit the fob ID

Reads four UID bytes, formats them as hexadecimal text, and sends the encoded ID while illuminating the scan LED.

while True:
    (stat, tag_type) = rdr.request(rdr.REQIDL)
    if stat == rdr.OK:
        (stat, raw_uid) = rdr.anticoll()
        if stat == rdr.OK:
            uid = "0x" + ''.join('{:02x}'.format(b) for b in raw_uid[:4])
            print("Card UID:", uid)
            led.value(1)
            print("LED ON")
            e.send(peer, uid.encode())
            time.sleep(1.5)
            led.value(0)

Pitch detection with the FFT

Collects 256 samples, removes the mean, finds the strongest frequency bin, and looks for a table entry within 5% of that frequency.

def get_condensed_fft():
    n_samples = 256
    samples = []
    for _ in range(n_samples):
        val = mic.value * (3.3 / 65536)
        samples.append(val)
        time.sleep(1 / 8000)
    data_array = np.array(samples, dtype=np.float)
    data_array = data_array - np.mean(data_array)
    real, imag = fft.fft(data_array)
    magnitude = np.sqrt(real**2 + imag**2)
    magnitude[0] = 0.0
    gc.collect()
    return magnitude[:len(magnitude) // 2].tolist()

def get_highest_frequency(fft_data, sampling_rate=8000):
    bin_size = sampling_rate / len(fft_data) / 2
    max_index = int(np.argmax(fft_data))
    return max_index * bin_size

def find_symbol_for_frequency(highest_frequency, tolerance=0.05):
    for entry in frequency_table:
        center_freq = entry['center_freq']
        if center_freq * (1 - tolerance) <= highest_frequency <= center_freq * (1 + tolerance):
            return entry['symbol']
    return "??"

Medium mode

Button 2 selects the medium exercise. Each target gets an eight-second listening window, with LED feedback after each one-second check.

if not button2.value:
    print('MEDIUM MODE:')
    for medium in medium_notes:
        print("\n🔊 Listening for:", medium)
        heard = False
        start_time = time.monotonic()
        while time.monotonic() - start_time < 8:
            second_start = time.monotonic()
            correct_this_second = False
            while time.monotonic() - second_start < 1.0:
                fft_data = get_condensed_fft()
                highest_freq = get_highest_frequency(fft_data)
                detected = find_symbol_for_frequency(highest_freq)
                if detected == medium:
                    correct_this_second = True
            if correct_this_second:
                print(f"  ✅ Heard {medium} this second!")
                heard = True
                led1.value = True
                time.sleep(0.2)
                led1.value = False
            else:
                print(f"  ❌ {medium} not heard this second.")
                led2.value = True
                time.sleep(0.2)
                led2.value = False
        if not heard:
            print(f"  ❌ {medium} not heard at all.")
